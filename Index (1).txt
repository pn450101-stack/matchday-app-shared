      match.clock.lastTickAt = Date.now();

      await save('match-current', match);
      checkClockMilestones();
    }, 60000);
  }

      await save('match-current', match);
      renderClockUI();

  // ---------- lineup ----------
  function classifyPosition(pos){
    const p = (pos||'').trim().toUpperCase();
    if(!p) return 'MID';
    const map = {
      GK:['GK','GOALKEEPER','KEEPER'],
      DEF:['CB','RB','LB','RWB','LWB','SW','DEF','DEFENDER','FULLBACK','FULL BACK','CENTRE BACK','CENTER BACK','RCB','LCB'],
      MID:['CM','CDM','CAM','DM','AM','MID','MIDFIELDER','RM','LM'],
      FWD:['ST','CF','FW','FORWARD','STRIKER','LW','RW','WINGER','ATT','ATTACKER']
    };
    for(const [group, list] of Object.entries(map)){
      if(list.includes(p)) return group;
    }
    if(p.includes('KEEP')) return 'GK';
    if(p.includes('BACK') || p.includes('DEF')) return 'DEF';
    if(p.includes('MID')) return 'MID';
    if(p.includes('FORWARD') || p.includes('STRIK') || p.includes('WING') || p.includes('ATT')) return 'FWD';
    return 'MID';
  }

  function buildLineupPostText(){
    const nf = getNextFixture();
    const opponent = nf ? nf.opponent : match.awayName;
    let dateLine = '';
    if(nf){
      const { sub } = formatFixtureDate(nf.date);
      dateLine = `${sub}${nf.time ? ' · ' + nf.time + ' KO' : ''}\n`;
    }
    const starters = match.lineup.starters;
    const groups = { FWD:[], MID:[], DEF:[], GK:[] };
    starters.forEach(p => groups[classifyPosition(p.pos)].push(p));
    const rows = ['FWD','MID','DEF','GK']
      .filter(g => groups[g].length)
      .map(g => `${g}\n${groups[g].map(p=>`${p.name} (#${p.num})`).join('   ·   ')}`)
      .join('\n\n');
    return `📋 TEAM NEWS\n${clubName} vs ${opponent}\n${dateLine}${match.lineup.formation ? 'Formation: ' + match.lineup.formation : ''}\n\n${rows || '—'}\n\n#${clubName.replace(/\s+/g,'')} #COYR`;
  }

  let tagClockMs = 0;
  let tagClockRunning = false;
  let tagClockInterval = null;

  function formatClockTime(ms){
    const totalSec = Math.floor(ms/1000);
    const m = String(Math.floor(totalSec/60)).padStart(2,'0');
    const s = String(totalSec%60).padStart(2,'0');
    return `${m}:${s}`;
  }

  function updateTagClockDisplay(){
    const el = $('tagClockDisplay');
    if(el) el.textContent = formatClockTime(tagClockMs);
  }

  function startTagClock(){
    if(tagClockInterval) return;
    tagClockRunning = true;
    const startedAt = Date.now() - tagClockMs;
    tagClockInterval = setInterval(()=>{
      tagClockMs = Date.now() - startedAt;
      updateTagClockDisplay();
    }, 250);
    const btn = $('tagClockToggle');
    if(btn) btn.textContent = '⏸ Pause';
  }

  function pauseTagClock(){
    tagClockRunning = false;
    if(tagClockInterval){ clearInterval(tagClockInterval); tagClockInterval = null; }
    const btn = $('tagClockToggle');
    if(btn) btn.textContent = '▶ Start';
  }

  function resetTagClock(){
    pauseTagClock();
    tagClockMs = 0;
    updateTagClockDisplay();
  }

  function goalsForPlayer(num){
    return match.events.filter(e => e.team !== 'opponent' && e.scorerNum === num).length;
  }

  function assistsForPlayer(num){
    return match.tags.filter(t => t.type === 'assist' && t.num === num).length;
  }

  function computeScore(p){
    const s = match.playerStats[p.num] || { rating:0 };
    return goalsForPlayer(p.num)*4 + assistsForPlayer(p.num)*3 + (s.rating||0);
  }

  async function normalizeMatch(){
    if(!match.lineup) match.lineup = { formation:'', starters:[] };
    if(!match.quotes) match.quotes = [];
    if(match.veoUrl === undefined) match.veoUrl = '';
    if(!match.playerStats) match.playerStats = {};
    if(match.motm === undefined) match.motm = null;
    if(!match.tags) match.tags = [];
    if(!match.clock) match.clock = { status:'not_started', running:false, lastTickAt:null, htShown:false, ftShown:false };
  }

  async function handleRemoteChange(key){
    if(key === 'match-current'){
      match = await load('match-current', match);
      await normalizeMatch();
      renderScore(); renderFeed(); renderLineup(); renderStats(); renderClockUI();
    } else if(key === 'squad'){
      squad = await load('squad', squad);
      renderSquad(); renderLineup();
    } else if(key === 'fixtures'){
      fixtures = await load('fixtures', fixtures);
      renderFixtures();
    } else if(key === 'club-name'){
      clubName = await load('club-name', clubName);
      $('clubNameInput').value = clubName;
    } else if(key === 'club-crest'){
      crestUrl = await load('club-crest', crestUrl);
      renderCrest();
    }
  }

  window.MatchdayApp = {
    init,
    subscribeRealtime(supabaseClient){
      supabaseClient
        .channel('app_data_changes')
        .on('postgres_changes', { event:'*', schema:'public', table:'app_data' }, (payload)=>{
          const key = payload.new ? payload.new.key : (payload.old ? payload.old.key : null);
          if(key) handleRemoteChange(key);
        })
        .subscribe();
    }
  };
})();
